{{/* tc-downloads shortcode Renders a gated download experience for Telegraf Controller. - Shows a "Download" button that opens a contact form modal. - After form submission (or a visit with ?ref=tc), JS renders the download links from the JSON data attribute. - The version comes from data/products.yml (telegraf_controller.latest_patch); the build matrix is defined below. URLs are constructed here at build time. - Every executable ships with a detached signature (.asc), and the release ships telegraf_controller-.digests (SHA-256 checksums) with its own .asc. */}} {{- $product := index .Site.Data.products "telegraf_controller" -}} {{- $baseURL := "https://dl.influxdata.com/telegraf-controller/releases" -}} {{- $version := $product.latest_patch -}} {{- $versionDir := printf "v%s" $version -}} {{/* Release build matrix. Each build is published as telegraf_controller--. Add a platform or an architecture here when the release workflow starts producing it. */}} {{- $matrix := slice (dict "name" "Linux" "builds" (slice (dict "os" "linux" "arch" "x64") (dict "os" "linux" "arch" "arm64"))) (dict "name" "macOS" "builds" (slice (dict "os" "macos" "arch" "x64") (dict "os" "macos" "arch" "arm64"))) (dict "name" "Windows" "builds" (slice (dict "os" "win" "arch" "x64" "ext" ".exe"))) -}} {{/* Build enriched platform data with constructed URLs */}} {{- $platforms := slice -}} {{- range $matrix -}} {{- $builds := slice -}} {{- range .builds -}} {{- $ext := .ext | default "" -}} {{- $filename := printf "telegraf_controller-%s-%s%s" .os .arch $ext -}} {{- $url := printf "%s/%s/%s" $baseURL $versionDir $filename -}} {{- $build := dict "arch" .arch "os" .os "filename" $filename "url" $url "signatureUrl" (printf "%s.asc" $url) -}} {{- $builds = $builds | append $build -}} {{- end -}} {{- $platforms = $platforms | append (dict "name" .name "builds" $builds) -}} {{- end -}} {{- $digestsFilename := printf "telegraf_controller-%s.digests" $version -}} {{- $digestsUrl := printf "%s/%s/%s" $baseURL $versionDir $digestsFilename -}} {{- $digests := dict "filename" $digestsFilename "url" $digestsUrl "signatureUrl" (printf "%s.asc" $digestsUrl) -}} {{- $enriched := dict "version" $version "platforms" $platforms "digests" $digests -}}
