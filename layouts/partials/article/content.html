{{- /* Render page content, resolving shared-content link placeholders. Shared files under content/shared/ are consumed by multiple products, so internal links are written as placeholders and resolved here against the consuming page's own content path. The path comes from data/products.yml via the product/version cascade -- it is never parsed out of the URL. Two placeholder forms are supported: //version/... Legacy. Works only for editions that share a product root (influxdb/v2 + influxdb/cloud, influxdb3/core + influxdb3/enterprise). Digit-insensitive on the root, so /influxdb/version/ also resolves on influxdb3/* pages. /product/version/... Neutral. Resolves for any product, including editions with different product roots (influxdb/v1 + enterprise_influxdb/v1). Prefer this in new shared content. */ -}} {{ if .Params.source }} {{ $contentPath := partial "product/content-path.html" . }} {{ $root := index (split $contentPath "/") 0 }} {{ $legacyRegex := print (replaceRE `[0-9]` "" $root) "(?:[0-9])*" "/version" }} {{/* Note: replace takes INPUT first, replaceRE takes INPUT last. Call both explicitly rather than piping, so the argument order stays obvious. */}} {{ $raw := readFile .Params.source }} {{ $raw = replace $raw "/product/version/" (print "/" $contentPath "/") }} {{ $raw = replaceRE $legacyRegex $contentPath $raw }} {{/* Tab-fragment-id collision detection (layouts/partials/tab-fragment-ids.html) needs the real markdown headings, which live in this sourced file, not in .RawContent -- a source: page's own body is just a stub. Store it before rendering so shortcodes invoked during RenderString can read it. */}} {{ .Page.Store.Set "tabFragmentRawContent" $raw }} {{ $raw | .RenderString | safeHTML }} {{ else }} {{ .Page.Store.Set "tabFragmentRawContent" .RawContent }} {{ .Content }} {{ end }}
