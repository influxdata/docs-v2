{{- /* Pairs tab links with their content sections by document order and injects a slugified id into each content section's opening tag. Params (dict): inner - the wrapper shortcode's already-rendered .Inner (string) page - the current Page context (for collision checks) labelClass - the div class wrapping the tab links (e.g. "tabs") contentClass - the section class wrapping tab content (e.g. "tab-content") */ -}} {{- $inner := .inner -}} {{- $page := .page -}} {{- $labelClass := .labelClass -}} {{- $contentClass := .contentClass -}} {{- $openTag := printf "
" $contentClass -}} {{- $parts := split $inner $openTag -}} {{- $labelsBlock := index $parts 0 -}} {{- $linkTags := findRE "[^<]*" $labelsBlock -}} {{- $out := $labelsBlock -}} {{- /* Collision candidates come from the page's raw markdown headings, not $page.Fragments or $page.Content: calling either from a shortcode that runs during this same page's content render creates a circular render dependency and deadlocks the build. */ -}} {{- $headingIDs := slice -}} {{- range findRE "(?m)^#{1,6}[ \t]+.+$" $page.RawContent -}} {{- $headingIDs = $headingIDs | append (anchorize (trim . "# \t")) -}} {{- end -}} {{- range $i, $part := after 1 $parts -}} {{- $slug := "" -}} {{- with index $linkTags $i -}} {{- $label := htmlUnescape (replaceRE "<[^>]+>" "" .) -}} {{- $base := anchorize $label -}} {{- $slug = $base -}} {{- $n := 2 -}} {{- range $headingIDs -}} {{- if eq . $slug -}} {{- $slug = printf "%s-%d" $base $n -}} {{- $n = add $n 1 -}} {{- end -}} {{- end -}} {{- $used := $page.Store.Get "tabFragmentIDs" -}} {{- if not $used -}}{{- $used = slice -}}{{- end -}} {{- range $used -}} {{- if eq . $slug -}} {{- $slug = printf "%s-%d" $base $n -}} {{- $n = add $n 1 -}} {{- end -}} {{- end -}} {{- $page.Store.Set "tabFragmentIDs" ($used | append $slug) -}} {{- end -}} {{- $out = printf "%s
%s" $out $contentClass $slug $part -}} {{- end -}} {{- return $out -}}
