{{- $valid := true -}} {{- range split .Inner "\n" -}} {{- $quotes := len (findRE `"` (. | htmlEscape)) -}} {{- if ne (mod $quotes 2) 0 -}} {{- $valid = false -}} {{- end -}} {{- end -}} {{- $code := .Inner | htmlEscape -}} {{- if $valid -}} {{- /* Apply replacements to escaped source, then mark only the generated markup safe. An unterminated string falls back to plain text. */ -}} {{- $code = replaceRE `([[:alnum:]_.-]+)::([^=,[:space:]]+)(=)` `$1::$2$3` $code -}} {{- $code = replaceRE `"[^"\n]*"` `$0` $code -}} {{- $code = replaceRE `(?m)^#.*$` `$0` $code -}} {{- end -}} {{- printf "
%s
" $code | safeHTML -}}
